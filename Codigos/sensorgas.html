Projeto Sensor de gás e fumaça

Código Arduino



#define Pino_Sensor 9
#define Pino_buzzer 8
#define Pino_LED 11

void setup() {
  pinMode(Pino_Sensor, INPUT);
  pinMode(Pino_LED, OUTPUT);
  pinMode(Pino_buzzer, OUTPUT);
}

void loop() {
  if (digitalRead(Pino_Sensor) == HIGH) {
    // Mantém ligados enquanto o sensor estiver em HIGH
    digitalWrite(Pino_LED, HIGH);
    tone(Pino_buzzer, 2000);
  }
  else {
    // Desliga quando o sensor voltar para LOW
    digitalWrite(Pino_LED, LOW);
    noTone(Pino_buzzer);
  }
}
