Projeto Seguidor de linha

Código Arduino



// =====================================================
// ROBÔ SEGUIDOR DE LINHA PRETA
// =====================================================
//
// A0 = sensor esquerdo
// A1 = sensor direito
//
// PRETO  = valor > 290
// BRANCO = valor <= 290
//
// PRETO + PRETO   -> frente
// PRETO + BRANCO  -> curva fechada esquerda
// BRANCO + PRETO  -> curva fechada direita
// BRANCO + BRANCO -> continua o último movimento
// =====================================================


// ===============================
// MOTOR ESQUERDO
// ===============================
#define IN1 10
#define IN2 9


// ===============================
// MOTOR DIREITO
// ===============================
#define IN3 6
#define IN4 5


// ===============================
// SENSORES
// ===============================
#define SENSOR_ESQUERDO A0
#define SENSOR_DIREITO A1


// ===============================
// CONFIGURAÇÕES
// ===============================
#define LIMITE 290

#define VELOCIDADE 75
#define VELOCIDADE_CURVA 75


// =====================================================
// ÚLTIMO MOVIMENTO
// =====================================================
//
// 0 = frente
// 1 = esquerda
// 2 = direita
//
int ultimoMovimento = 0;


// =====================================================
// MOTOR ESQUERDO PARA FRENTE
// =====================================================

void motorEsquerdo(int velocidade) {

  analogWrite(IN1, velocidade);
  digitalWrite(IN2, LOW);
}


// =====================================================
// MOTOR DIREITO PARA FRENTE
// =====================================================

void motorDireito(int velocidade) {

  digitalWrite(IN3, LOW);
  analogWrite(IN4, velocidade);
}


// =====================================================
// PARAR ESQUERDO
// =====================================================

void pararEsquerdo() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
}


// =====================================================
// PARAR DIREITO
// =====================================================

void pararDireito() {

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


// =====================================================
// PARAR OS DOIS
// =====================================================

void pararMotores() {

  pararEsquerdo();
  pararDireito();
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(9600);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(SENSOR_ESQUERDO, INPUT);
  pinMode(SENSOR_DIREITO, INPUT);

  pararMotores();

  Serial.println("SEGUIDOR DE LINHA INICIADO");
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  int A0valor = analogRead(SENSOR_ESQUERDO);
  int A1valor = analogRead(SENSOR_DIREITO);

  bool pretoEsquerdo = A0valor > LIMITE;
  bool pretoDireito  = A1valor > LIMITE;


  // ===================================================
  // PRETO + PRETO
  // FRENTE
  // ===================================================

  if (pretoEsquerdo && pretoDireito) {

    motorEsquerdo(VELOCIDADE);
    motorDireito(VELOCIDADE);

    // Guarda que o último movimento foi para frente
    ultimoMovimento = 0;

    Serial.println("PRETO + PRETO -> FRENTE");
  }


  // ===================================================
  // PRETO + BRANCO
  // CURVA FECHADA PARA ESQUERDA
  // ===================================================

  else if (pretoEsquerdo && !pretoDireito) {

    pararEsquerdo();
    motorDireito(VELOCIDADE_CURVA);

    // Guarda a última direção
    ultimoMovimento = 1;

    Serial.println("PRETO + BRANCO -> ESQUERDA");
  }


  // ===================================================
  // BRANCO + PRETO
  // CURVA FECHADA PARA DIREITA
  // ===================================================

  else if (!pretoEsquerdo && pretoDireito) {

    motorEsquerdo(VELOCIDADE_CURVA);
    pararDireito();

    // Guarda a última direção
    ultimoMovimento = 2;

    Serial.println("BRANCO + PRETO -> DIREITA");
  }


  // ===================================================
  // BRANCO + BRANCO
  // PERDEU A LINHA
  //
  // CONTINUA O ÚLTIMO MOVIMENTO
  // ===================================================

  else {

    if (ultimoMovimento == 0) {

      // Último movimento foi para frente
      motorEsquerdo(VELOCIDADE);
      motorDireito(VELOCIDADE);

      Serial.println("LINHA PERDIDA -> CONTINUANDO FRENTE");
    }

    else if (ultimoMovimento == 1) {

      // Último movimento foi esquerda
      pararEsquerdo();
      motorDireito(VELOCIDADE_CURVA);

      Serial.println("LINHA PERDIDA -> CONTINUANDO ESQUERDA");
    }

    else if (ultimoMovimento == 2) {

      // Último movimento foi direita
      motorEsquerdo(VELOCIDADE_CURVA);
      pararDireito();

      Serial.println("LINHA PERDIDA -> CONTINUANDO DIREITA");
    }
  }


  // ===================================================
  // MONITOR SERIAL
  // ===================================================

  Serial.print("A0 = ");
  Serial.print(A0valor);

  Serial.print(" | A1 = ");
  Serial.print(A1valor);

  Serial.print(" | E = ");
  Serial.print(pretoEsquerdo ? "PRETO" : "BRANCO");

  Serial.print(" | D = ");
  Serial.println(pretoDireito ? "PRETO" : "BRANCO");

  delay(20);
}
