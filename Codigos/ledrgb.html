Projeto LED RGB

Código Arduino



// LED RGB
const int redPin = 9;
const int bluePin = 11;

// Potenciômetro
const int potPin = A0;

// Botão
const int buttonPin = 2;

bool modo = false; // false = vermelho, true = azul
bool ultimoEstado = HIGH;

void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(bluePin, OUTPUT);

  pinMode(buttonPin, INPUT_PULLUP);
}

void loop() {

  // leitura do botão
  bool estadoAtual = digitalRead(buttonPin);

  // detecta clique (HIGH -> LOW)
  if (ultimoEstado == HIGH && estadoAtual == LOW) {
    modo = !modo;
    delay(50); // debounce
  }

  ultimoEstado = estadoAtual;

  // brilho do potenciômetro
  int brilho = map(analogRead(potPin), 0, 1023, 0, 255);

  if (!modo) {
    // 🔴 perigo
    analogWrite(redPin, brilho);
    analogWrite(bluePin, 0);
  } 
  else {
    // 🔵 saída segura
    analogWrite(redPin, 0);
    analogWrite(bluePin, brilho);
  }

  delay(10);
}
